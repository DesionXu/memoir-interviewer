// ===== 分页处理：每章另起一页，每页底部加页码 =====
function paginateMemoir(text, charsPerPage = 500) {
    const blocks = text.split(/\n+/).map(b => b.trim()).filter(Boolean);
    const pages = [];
    let page = [];
    let len = 0;
    let inToc = false;   // 目录里的"第X章"不是正文，不另起页

    const flush = () => {
        if (page.length) {
            pages.push(page);
            page = [];
            len = 0;
        }
    };

    for (const block of blocks) {
        const isSection = /^【.+】/.test(block);
        if (isSection) {
            flush();
            inToc = block.includes('目录');
        }
        const isChapter = !inToc && /^第[一二三四五六七八九十百0-9０-９]+章/.test(block);
        if (isChapter) flush();

        page.push(block);
        len += block.length;
        if (len >= charsPerPage) flush();
    }
    flush();

    return pages.map((p, i) =>
        p.join('\n\n') + '\n\n— 第 ' + (i + 1) + ' 页 —'
    ).join('\n\n\n');
}

/**

 * API Key 从本机设置读取，网页端无法接触到。
 */
export async function callChatApi(messages: ChatMessage[], maxTokens: number, temperature: number): Promise<string> {
  const apiKey = await SettingsService.getApiKey();
  if (!apiKey || apiKey === AppConfig.DEFAULT_API_KEY || apiKey === '') {
    throw new Error('请先在"设置"页填写 DeepSeek API Key');
  }

  const request = http.createHttp();
  try {

第 37 页
    const response = await request.request(AppConfig.API_URL, {
      method: http.RequestMethod.POST,
      header: {
        'Content-Type': 'application/json',
        'Authorization': 'Bearer ' + apiKey
      },
      extraData: JSON.stringify({
        model: AppConfig.MODEL,
        messages: messages,
        stream: false,
        max_tokens: maxTokens,
        temperature: temperature
      }),
      expectDataType: http.HttpDataType.STRING,
      connectTimeout: 60000,
      readTimeout: 180000
    });
    if (response.responseCode !== 200) {
      hilog.error(DOMAIN, 'ApiService', 'DeepSeek API error: %{public}d', response.responseCode);
      throw new Error('API 返回错误码 ' + response.responseCode);
    }
    return response.result as string;
  } catch (err) {
    const e = err as BusinessError;
    hilog.error(DOMAIN, 'ApiService', 'request failed: %{public}s', e.message);
    throw new Error('请求失败：' + e.message);
  } finally {

  }

import { preferences } from '@kit.ArkData';
import { common } from '@kit.AbilityKit';
import { BusinessError } from '@kit.BasicServicesKit';
import { hilog } from '@kit.PerformanceAnalysisKit';
import { AppConfig } from './Config';

const DOMAIN: number = 0x0000;
const STORE_NAME: string = 'memoir_settings';
const KEY_API_KEY: string = 'api_key';


第 41 页
/**
 * 本机设置存储（preferences）：保存 DeepSeek API Key，仅存本机，不上传。
 */
export class SettingsService {
  private static store: preferences.Preferences | null = null;

  static async init(context: common.Context): Promise<void> {
    if (SettingsService.store) {
      return;
    }
    try {
      SettingsService.store = await preferences.getPreferences(context, STORE_NAME);
    } catch (err) {
      const e = err as BusinessError;
      hilog.error(DOMAIN, 'SettingsService', 'init failed: %{public}s', e.message);
    }
  }

  static async getApiKey(): Promise<string> {
    if (!SettingsService.store) {
      return AppConfig.DEFAULT_API_KEY;
    }
    try {
      const value = await SettingsService.store.get(KEY_API_KEY, AppConfig.DEFAULT_API_KEY);
      return value as string;
    } catch (err) {
      return AppConfig.DEFAULT_API_KEY;
    }
  }

  static async setApiKey(key: string): Promise<void> {
    if (!SettingsService.store) {
      return;
    }
    try {
      await SettingsService.store.put(KEY_API_KEY, key);
      await SettingsService.store.flush();
    } catch (err) {
      const e = err as BusinessError;
      hilog.error(DOMAIN, 'SettingsService', 'setApiKey failed: %{public}s', e.message);
    }
  }
}

 */
export async function ensureMicPermission(context: common.Context): Promise<boolean> {
  const atManager = abilityAccessCtrl.createAtManager();
  try {
    const result = await atManager.requestPermissionsFromUser(context, [Permissions.MICROPHONE]);
    return result.authResults.length > 0 && result.authResults[0] === 0;
  } catch (err) {
    const e = err as BusinessError;
    return false;
  }
}
